# Build the container image.
#[user@desktop]

    buildtag=$(date '+%Y.%m.%d')
    buildtime=$(date '+%Y-%m-%dT%H:%M:%S')

    source "${HOME:?}/lithosia.env"
    pushd  "${LITHOSIA_CODE}"

        podman build \
            --no-cache \
            --build-arg "buildtag=${buildtag:?}" \
            --build-arg "buildtime=${buildtime:?}" \
            --tag "zarquan/lithosia-quadra:latest" \
            --tag "zarquan/lithosia-quadra:${buildtag:?}" \
            docker

    popd

# -----------------------------------------------------
# Launch the container with our projects mounted as volumes.
#[user@desktop]

    source "${HOME:?}/lithosia.env"
    source "${HOME:?}/calycopis.env"
    source "${CALYCOPIS_CODE}/bin/container-host.sh"

    podman network create \
        --ignore \
        "lithosia-dsh-net"

    podman pod create \
        --replace \
        --name    "lithosia-dsh-pod" \
        --network "lithosia-dsh-net" \
        --publish 3081:3081

    podman run \
        --rm \
        --tty \
        --interactive \
        --pod    "lithosia-dsh-pod" \
        --name   "lithosia-dsh-container" \
        --env    "CONTAINER_HOST=unix:///run/podman/podman.sock" \
        --volume "${HOST_CONTAINER_PATH}:/run/podman/podman.sock:rw,Z" \
        --env    "CALYCOPIS_CODE=/Calycopis/Calycopis-broker" \
        --volume "${CALYCOPIS_CODE:?}:/Calycopis/Calycopis-broker:rw,Z" \
        --env    "TREBULA_CODE=/Calycopis/Calycopis-openapi" \
        --volume "${TREBULA_CODE:?}:/Calycopis/Calycopis-openapi:rw,Z" \
        --env    "LITHOSIA_CODE=/Zarquan/lithosia-quadra" \
        --volume "${LITHOSIA_CODE:?}:/Zarquan/lithosia-quadra:rw,Z" \
        --env    "DSH_HOME=/Zarquan/lithosia-quadra/dsh" \
        --secret 'dsh-github-token' \
        localhost/zarquan/lithosia-quadra:latest \
            bash
